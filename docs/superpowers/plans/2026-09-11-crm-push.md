# Automatic CRM Push Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver every website lead to the ZOLVEX CRM automatically, with retries, backoff and an audited manual re-push, without ever blocking the public enquiry form.

**Architecture:** The `Enquiry` table doubles as the outbox. A fire-and-forget nudge after insert plus a 30-second in-process worker claim rows with `FOR UPDATE SKIP LOCKED`, map them through a single adapter module, and POST them to the CRM. The CRM's wire format lives in exactly one file (`mapper.ts`), so the whole pipeline can merge and ship disabled before that format is known.

**Tech Stack:** Node 20, TypeScript 5.6 (ESM), Express 4.21, Prisma 6 / PostgreSQL, Zod 4, Vitest + Supertest, Next.js 16 (admin + public UI).

**Spec:** `docs/superpowers/specs/2026-09-11-crm-push-design.md`

## Global Constraints

- **ESM imports:** every relative import ends in `.js` (e.g. `import { prisma } from "../../db/prisma.js"`). This codebase is ESM; omitting the extension breaks the build.
- **Tests run against a real Postgres test database** via `DATABASE_URL_TEST` (see `src/test-setup.ts`). Run them with `npm test --workspace=apps/api`.
- **No network I/O in tests.** The CRM transport is injected; tests always pass a fake.
- **Migrations are additive only.** Every new column is nullable. No backfill.
- **`writeAuditRow(tx, …)` must be called inside a `prisma.$transaction`** — never standalone. Its signature is `writeAuditRow(tx, { entity, entityId, action, actorId, ipAddress?, diff })`.
- **Admin routes mount at `/admin/api/...`**; public routes at `/api/...`.
- **Authenticated admin identity is `req.actor` (`{ id, role }`)** on an `AuthedRequest`, set by `requireAuth`.
- **Backoff schedule:** `[1, 5, 15, 60, 360]` minutes, `MAX_ATTEMPTS = 6`.
- **Stale claim threshold:** 5 minutes.
- **`CRM_ENABLED` unset ⇒ the worker never starts.** This plan must merge safely with the CRM disabled.
- **Status meanings:** `new` = never attempted · `failed` = retry scheduled · `needs_manual_push` = terminal · `pushed_to_crm` = done.

```ts
import type { Enquiry } from "@prisma/client";

/**
 * The single place that knows what the CRM calls each field.
 *
 * Until the ZOLVEX CRM team supplies their request format this emits our own
 * field names, which is what the admin payload preview renders — and what we
 * send them as a concrete sample. When their spec lands, this function is the
 * only thing that has to change.
 */
export type CrmLeadPayload = Record<string, unknown>;

export function toCrmLead(enquiry: Enquiry): CrmLeadPayload {
  const payload: CrmLeadPayload = {
    // Our cuid, so the CRM (and we) can reconcile a lead across retries.
    reference: enquiry.id,
    name: enquiry.name,
    phone: enquiry.phone,
    place: enquiry.place,
    service: enquiry.serviceName,
    submittedAt: enquiry.createdAt.toISOString(),
  };

  // Absent is different from empty: send nothing rather than a null the CRM
  // would have to interpret.
  if (enquiry.email) payload.email = enquiry.email;
  if (enquiry.message) payload.message = enquiry.message;
  if (enquiry.preferredDate) {
    payload.preferredDate = enquiry.preferredDate.toISOString().slice(0, 10);
  }

  return payload;
}
```

```ts
export class CrmNotConfiguredError extends Error {}

/** Credentials are set but the CRM rejected them. Terminal — retrying will not help. */
export class CrmAuthError extends Error {}

/** The CRM understood the request and refused the payload. Terminal. */
export class CrmRejectedError extends Error {}

/** Transport failure, timeout, throttling or a CRM-side fault. Retryable. */
export class CrmUnavailableError extends Error {}
```

- [ ] **Step 4: Write `config.ts`**

```ts
import { CrmNotConfiguredError } from "./errors.js";

/**
 * Whether the dispatcher should run at all. Unset means the pipeline is built
 * but dormant, which is how this ships before the CRM spec exists.
 */
export function isCrmEnabled(): boolean {
  const flag = process.env.CRM_ENABLED;
  return flag === "1" || flag?.toLowerCase() === "true";
}

export function getCrmConfig(): { url: string; apiKey: string; timeoutMs: number } {
  const url = process.env.CRM_API_URL;
  const apiKey = process.env.CRM_API_KEY;
  if (!url || !apiKey) {
    throw new CrmNotConfiguredError("CRM env vars (CRM_API_URL/CRM_API_KEY) are not set");
  }
  const timeoutMs = process.env.CRM_TIMEOUT_MS ? Number(process.env.CRM_TIMEOUT_MS) : 10_000;
  return { url, apiKey, timeoutMs };
}
```

- [ ] **Step 5: Write `client.ts`**

```ts
import { getCrmConfig } from "./config.js";
import { CrmAuthError, CrmRejectedError, CrmUnavailableError } from "./errors.js";
import type { CrmLeadPayload } from "./mapper.js";

/** Injected so tests never open a socket. */
export type CrmTransport = (url: string, init: RequestInit) => Promise<Response>;

const defaultTransport: CrmTransport = (url, init) => fetch(url, init);

/**
 * 401/403 and 400/422 are deliberately terminal. Wrong credentials and a
 * refused payload do not heal on their own; retrying them burns rate limit
 * and disguises a configuration fault as a transient one.
 */
function classify(status: number, body: string): Error {
  if (status === 401 || status === 403) {
    return new CrmAuthError(`CRM rejected our credentials (${status})`);
  }
  if (status === 408 || status === 429 || status >= 500) {
    return new CrmUnavailableError(`CRM unavailable (${status}) ${body}`.trim());
  }
  return new CrmRejectedError(`CRM refused the payload (${status}) ${body}`.trim());
}

export async function sendLead(
  payload: CrmLeadPayload,
  idempotencyKey: string,
  transport: CrmTransport = defaultTransport
): Promise<unknown> {
  // Throws CrmNotConfiguredError before any I/O when the deployment has no
  // credentials — the caller treats that as terminal, not retryable.
  const { url, apiKey, timeoutMs } = getCrmConfig();

  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);

  let response: Response;
  try {
    response = await transport(url, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Authorization: `Bearer ${apiKey}`,
        // Our cuid. A retry after an ambiguous timeout must not create a
        // second lead — pending confirmation that the CRM honours this.
        "Idempotency-Key": idempotencyKey,
      },
      body: JSON.stringify(payload),
      signal: controller.signal,
    });
  } catch (error) {
    throw new CrmUnavailableError(`CRM request failed: ${(error as Error).message}`);
  } finally {
    clearTimeout(timer);
  }

  const text = await response.text();
  if (!response.ok) throw classify(response.status, text);

  try {
    return text ? JSON.parse(text) : {};
  } catch {
    // A 2xx with a non-JSON body still means accepted; keep the raw text.
    return { raw: text };
  }
}
```

```ts
/**
 * Six attempts across roughly seven and a quarter hours, then a human takes
 * over. Indexed by attemptCount *after* the failure has been recorded, so
 * attempt 1 waits a minute and attempt 6 gives up.
 */
const BACKOFF_MINUTES = [1, 5, 15, 60, 360];

export const MAX_ATTEMPTS = 6;

export function nextAttemptDelayMs(attemptCount: number): number | null {
  if (attemptCount >= MAX_ATTEMPTS) return null;
  const minutes = BACKOFF_MINUTES[attemptCount - 1];
  if (minutes === undefined) return null;
  return minutes * 60_000;
}
```

```ts
import type { Enquiry, EnquiryStatus } from "@prisma/client";
import { prisma } from "../../db/prisma.js";
import { toCrmLead } from "./mapper.js";
import { sendLead, type CrmTransport } from "./client.js";
import { CrmUnavailableError } from "./errors.js";
import { nextAttemptDelayMs } from "./backoff.js";

const STALE_CLAIM_MINUTES = 5;
const DEFAULT_BATCH = 20;

/**
 * Claim work atomically. SKIP LOCKED means a second instance — or a manual
 * push racing a tick — can never pick up the same lead twice. A claim older
 * than STALE_CLAIM_MINUTES is reclaimable so a process that died mid-flight
 * cannot strand a row forever.
 */
export async function claimDueEnquiries(limit: number = DEFAULT_BATCH): Promise<Enquiry[]> {
  return prisma.$queryRaw<Enquiry[]>`
    UPDATE "Enquiry" SET "claimedAt" = NOW()
    WHERE id IN (
      SELECT id FROM "Enquiry"
      WHERE status IN ('new', 'failed')
        AND ("nextAttemptAt" IS NULL OR "nextAttemptAt" <= NOW())
        AND ("claimedAt" IS NULL
             OR "claimedAt" < NOW() - (${STALE_CLAIM_MINUTES} * INTERVAL '1 minute'))
      ORDER BY "createdAt" ASC
      LIMIT ${limit}
      FOR UPDATE SKIP LOCKED
    )
    RETURNING *;
  `;
}

export async function dispatchOne(
  enquiry: Enquiry,
  transport?: CrmTransport
): Promise<EnquiryStatus> {
  try {
    const response = await sendLead(toCrmLead(enquiry), enquiry.id, transport);
    await prisma.enquiry.update({
      where: { id: enquiry.id },
      data: {
        status: "pushed_to_crm",
        crmResponse: response as never,
        claimedAt: null,
        lastError: null,
        nextAttemptAt: null,
      },
    });
    return "pushed_to_crm";
  } catch (error) {
    const attemptCount = enquiry.attemptCount + 1;
    const message = (error as Error).message.slice(0, 500);

    // Only CrmUnavailableError is worth trying again. A refused payload, bad
    // credentials or a missing configuration all need a human.
    const retryable = error instanceof CrmUnavailableError;
    const delayMs = retryable ? nextAttemptDelayMs(attemptCount) : null;

    const status: EnquiryStatus = delayMs === null ? "needs_manual_push" : "failed";
    await prisma.enquiry.update({
      where: { id: enquiry.id },
      data: {
        status,
        attemptCount,
        lastError: message,
        claimedAt: null,
        nextAttemptAt: delayMs === null ? null : new Date(Date.now() + delayMs),
      },
    });
    return status;
  }
}

export async function drainOnce(transport?: CrmTransport): Promise<number> {
  const claimed = await claimDueEnquiries();
  for (const enquiry of claimed) {
    await dispatchOne(enquiry, transport);
  }
  return claimed.length;
}
```

```ts
import { isCrmEnabled } from "./config.js";
import { drainOnce } from "./dispatcher.js";
import type { CrmTransport } from "./client.js";

const DEFAULT_INTERVAL_MS = 30_000;

let timer: NodeJS.Timeout | null = null;
let inFlight = false;

export function isWorkerRunning(): boolean {
  return timer !== null;
}

/**
 * One pass. The in-flight guard means a slow drain can never overlap the next
 * interval — the queue is drained by whichever tick is already running.
 */
export async function tick(transport?: CrmTransport): Promise<void> {
  if (inFlight) return;
  inFlight = true;
  try {
    await drainOnce(transport);
  } catch (error) {
    // A worker must never take the process down. Per-row failures are already
    // recorded on the row itself; this only catches infrastructure faults.
    console.error("crm worker tick failed:", (error as Error).message);
  } finally {
    inFlight = false;
  }
}

/**
 * Fire-and-forget. Called after an enquiry is created so the common case does
 * not wait up to a full interval. Never awaited, never allowed to reject into
 * the request path.
 */
export function nudge(): void {
  void tick().catch(() => {});
}

export function startCrmWorker(intervalMs: number = DEFAULT_INTERVAL_MS): void {
  if (!isCrmEnabled()) return;
  if (timer) return;
  timer = setInterval(() => void tick(), intervalMs);
  // Do not hold the process open on this timer alone.
  timer.unref?.();
}

export function stopCrmWorker(): void {
  if (!timer) return;
  clearInterval(timer);
  timer = null;
}
```

```ts
import type { AuthedRequest } from "../../lib/auth/middleware.js";
import { toCrmLead } from "../../lib/crm/mapper.js";
import { dispatchOne } from "../../lib/crm/dispatcher.js";
import { writeAuditRow } from "../../lib/services/audit.js";

/**
 * The only route out of `needs_manual_push`. Resets the retry budget first so
 * a lead rescued by hand gets a full set of attempts rather than failing
 * straight back to terminal on its next hiccup.
 */
export async function pushOne(req: AuthedRequest, res: Response) {
  const existing = await prisma.enquiry.findUnique({ where: { id: req.params.id } });
  if (!existing) {
    res.status(404).json({ error: "not_found" });
    return;
  }

  const reset = await prisma.enquiry.update({
    where: { id: existing.id },
    data: { attemptCount: 0, nextAttemptAt: null, lastError: null, claimedAt: null },
  });

  const status = await dispatchOne(reset);

  await prisma.$transaction(async (tx) => {
    await writeAuditRow(tx, {
      entity: "enquiry",
      entityId: existing.id,
      action: "push",
      actorId: req.actor!.id,
      diff: { status: { old: existing.status, new: status } },
    });
  });

  res.status(200).json({ id: existing.id, status });
}

  if (!enquiry) {

    return;
```
